async function fetchUsers() { try { const response = await fetch('https://jsonplaceholder.typicode.com/users'); if (!response.ok) { throw new Error('Ошибка сети: ${response.status}'); } const users = await response.json(); users.forEach(user => { console.log(user.name); }); } catch (error) { console.error('Произошла ошибка при получении данных:', error.message); } } fetchUsers();
