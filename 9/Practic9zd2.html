async function createPost(title, body, userId) { try { const postData = { title: title, body: body, userId: userId }; const response = await axios.post( 'https://jsonplaceholder.typicode.com/posts', postData ); console.log('Пост создан:', response.data); return response.data; // Возвращаем данные созданного поста } catch (error) { console.error('Ошибка при создании поста:', error); throw error; // Перебрасываем ошибку для дальнейшей обработки } }
